\begin{pygmented}[lang=bash]
function main {
  PRINT_VERSION=false
  declare -a IGNORE_LIST

  GETOPT_OUTPUT=$(\
    getopt -o     'vi:' \
           --long 'version,ignore:' \
           --     "$@")

  eval set -- "$GETOPT_OUTPUT"
  unset GETOPT_OUTPUT

  while true; do
    case "$1" in
      '-v'|'--version')
        PRINT_VERSION=true
        shift
        continue
      ;;
      '-i'|'--ignore')
        IGNORE_LIST+=("$2")
        shift 2
        continue
      ;;
      '--')
        shift
        break
      ;;
    esac
  done

  if [ "$PRINT_VERSION" = true ]; then
    echo "Version 0.1"
  else
    echo "${IGNORE_LIST[*]}"
  fi
}
\end{pygmented}
